@startuml


skinparam backgroundColor white
skinparam shadowing false
skinparam defaultFontName "Microsoft YaHei"
skinparam sequence {
    ArrowColor black
    LifeLineBorderColor black
    LifeLineBackgroundColor white
    ParticipantBorderColor black
    ParticipantBackgroundColor white
}

actor "系统启动器" as Launcher
participant "create_app()\n应用工厂" as Factory
participant "FastAPI应用\n食智眸" as App
participant "init_data()\n初始化入口" as InitData
participant "数据库迁移\ninit_db()" as InitDB
participant "模型加载\ninit_yolo_model()" as InitYolo
participant "基础数据初始化" as InitBase
participant "食物数据初始化\ninit_food_data()" as InitFood

Launcher -> Factory : 调用 create_app()
activate Factory

Factory -> App : 创建应用实例
Factory -> App : 配置标题、CORS、异常处理
Factory -> App : 挂载静态文件
Factory -> App : 注册 API 路由前缀 /api

Factory -> InitData : 启动时执行初始化
activate InitData

InitData -> InitDB : 初始化数据库与迁移
activate InitDB
InitDB -> InitDB : 创建数据库并执行 Aerich 迁移
alt 迁移历史异常
    InitDB -> InitDB : 重建迁移目录
end
InitDB --> InitData : 数据库就绪
deactivate InitDB

InitData -> InitYolo : 加载 YOLOv11 模型
activate InitYolo
InitYolo -> InitYolo : 优先加载 best.pt
alt best.pt 不存在
    InitYolo -> InitYolo : 回退加载 last.pt
end
InitYolo -> InitYolo : 模型预热
InitYolo --> InitData : 模型就绪
deactivate InitYolo

InitData -> InitBase : 初始化系统基础数据
activate InitBase
InitBase -> InitBase : 创建超级管理员
InitBase -> InitBase : 初始化系统菜单
InitBase -> InitBase : 同步 API 权限
InitBase -> InitBase : 创建基础角色并分配权限
InitBase --> InitData : 基础数据就绪
deactivate InitBase

InitData -> InitFood : 初始化食物与营养数据
activate InitFood
InitFood -> InitFood : 检查 FoodCategory 表
alt 表为空
    InitFood -> InitFood : 导入15种食物类别及营养数据
end
InitFood --> InitData : 食物数据就绪
deactivate InitFood

InitData --> Factory : 初始化完成
deactivate InitData

Factory --> Launcher : 返回 FastAPI 应用实例
deactivate Factory

@enduml
