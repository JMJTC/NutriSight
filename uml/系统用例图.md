    usecase "维护个人信息" as UC_Profile

    usecase "食物图像识别" as UC_Recognize

    usecase "营养成分匹配" as UC_MatchNutrition
    usecase "生成营养分析" as UC_Analysis
    usecase "生成饮食建议" as UC_Recommend
  }
  
  package "用户功能" {
    usecase "查看识别历史" as UC_History
    usecase "查看营养可视化" as UC_Chart
  }
  
  package "系统管理" {
    usecase "用户管理" as UC_UserManage
    usecase "角色权限管理" as UC_RBAC

    usecase "接口权限管理" as UC_API
    usecase "操作日志查看" as UC_Log
