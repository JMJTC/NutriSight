@startuml


skinparam backgroundColor white
skinparam shadowing false
skinparam defaultFontName "Microsoft YaHei"
skinparam activity {
    BackgroundColor white
    BorderColor black
    FontColor black
}
skinparam ArrowColor black

start

:调用 predict(image_path);

if (模型已加载?) then (否)
    :进入 _load_lock;
    :查找权重文件\nbest.pt 优先，last.pt 兜底;

    if (权重存在并加载成功?) then (是)
        :保存 _model;\n设置 _model_loaded=True;
    else (否)
        :记录 _load_error;
        stop
    endif
else (是)
    :复用已加载模型;
endif

:检查图片路径;

if (图片存在?) then (是)
    :进入 _predict_lock;
    :执行 model.predict(conf=0.25);
else (否)
    :返回路径错误;
    stop
endif

if (检测输出存在?) then (是)
    :解析检测结果\n类别、置信度、bbox;
else (否)
    :解析分类结果\nTop-3或Top-1类别;
endif

:返回结构化预测结果;

stop

@enduml
