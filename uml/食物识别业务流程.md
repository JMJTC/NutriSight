@startuml


skinparam backgroundColor white
skinparam shadowing false
skinparam defaultFontName "Microsoft YaHei"
skinparam sequence {
    ArrowColor black
    LifeLineBorderColor black
    LifeLineBackgroundColor white
    ParticipantBorderColor black
    ParticipantBackgroundColor white
}

actor "用户" as User
participant "Vue3前端" as Web
participant "API路由层\n/recognize" as API
participant "FoodController\n控制层" as Controller
participant "FoodRecognitionService\n识别服务层" as Service
participant "YoloService\n模型推理层" as Yolo
database "MySQL数据库" as DB
collections "uploads目录" as Uploads

User -> Web : 上传食物图片
Web -> API : multipart/form-data\n携带token
API -> API : DependAuth认证

alt 未认证
    API --> Web : 401未认证
else 已认证
    API -> Controller : 转发识别请求
    Controller -> Service : recognize_food(file, user_id)

    group 图片校验
        Service -> Service : 校验扩展名、MIME、大小、内容、分辨率
    end

    alt 图片非法
        Service --> Controller : 返回校验失败
        Controller --> API : 错误响应
        API --> Web : 400图片不合法
    else 校验通过
        group 文件持久化
            Service -> Uploads : UUID命名保存图片
        end

        group 模型推理
            Service -> Yolo : asyncio.to_thread()\npredict_with_annotation()
            Yolo -> Uploads : 生成标注图片
            Yolo --> Service : 返回class_id、置信度、标注图路径
        end

        group 结果处理与数据存储
            loop 遍历预测结果
                Service -> DB : 匹配FoodCategory
                Service -> DB : 查询Nutrition
                Service -> DB : 创建RecognitionDetail
            end
            Service -> Service : 汇总营养成分\n计算热量等级
            Service -> DB : 创建NutritionAnalysis\nRecognitionRecord
        end

        Service --> Controller : 返回结构化JSON结果
        Controller --> API : 封装接口响应
        API --> Web : 返回识别结果、营养信息和标注图
    end
end

Web --> User : 展示识别结果

@enduml
